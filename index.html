โค้ด Google Apps Script (คัดลอกนำไปวางใน Apps Script): 
function doGet(e) {
  return ContentService.createTextOutput("ระบบบันทึกคะแนนโปรเจกไทล์พร้อมใช้งาน (Active)")
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "เวลาบันทึก", 
        "ชื่อ-นามสกุล", 
        "ชั้น", 
        "เลขที่", 
        "ข้อ 1", 
        "ข้อ 2", 
        "ข้อ 3", 
        "ข้อ 4", 
        "ข้อ 5", 
        "คะแนนรวม (100)"
      ]);
    }
    
    var data;
    if (e && e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      data = e.parameter;
    } else {
      throw new Error("No data received");
    }
    
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString("th-TH"),
      data.studentName || "-",
      data.studentClass || "-",
      data.studentNo || "-",
      data.q1 !== undefined ? data.q1 : 0,
      data.q2 !== undefined ? data.q2 : 0,
      data.q3 !== undefined ? data.q3 : 0,
      data.q4 !== undefined ? data.q4 : 0,
      data.q5 !== undefined ? data.q5 : 0,
      data.totalScore !== undefined ? data.totalScore : 0
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "status": "success" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

สรุปสูตรฟิสิกส์: การเคลื่อนที่แบบโปรเจกไทล์

ใช้เป็นคู่มือคำนวณหามุมยิง ($\theta$) และความเร็วต้น ($v_0$)

1. การแยกองค์ประกอบความเร็วต้น ($v_0$)
$v_x = v_0 \cdot \cos\theta$ (ความเร็วแนวราบ คงที่)
$v_y = v_0 \cdot \sin\theta$ (ความเร็วแนวดิ่ง เปลี่ยนตาม $g$)
2. สมการตำแหน่งตามเวลา ($t$)
แนวราบ: $X = (v_0 \cos\theta) \cdot t$  $\rightarrow$  $t = \frac{X}{v_0 \cos\theta}$
แนวดิ่ง: $Y = h_0 + (v_0 \sin\theta) \cdot t - \frac{1}{2}g t^2$
3. ระยะตกแนวราบ ($R$) กรณีพื้นราบ ($h_0 = 0, Y = 0$)
เวลาลอยทั้งหมด: $T = \frac{2 v_0 \sin\theta}{g}$
ระยะตกแนวราบ: $R = \frac{v_0^2 \sin(2\theta)}{g}$
